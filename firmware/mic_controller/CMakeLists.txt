cmake_minimum_required(VERSION 3.13)
set(PICO_PLATFORM rp2350-arm-s)
set(PICO_BOARD fc_magicon)
set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR}/boards)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(mic_controller C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(mic_controller main.c)
target_include_directories(mic_controller PRIVATE ${CMAKE_CURRENT_LIST_DIR})
pico_generate_pio_header(mic_controller ${CMAKE_CURRENT_LIST_DIR}/cpu_m2.pio)
target_link_libraries(mic_controller pico_stdlib pico_multicore hardware_pio hardware_clocks hardware_watchdog tinyusb_host)
target_compile_definitions(mic_controller PRIVATE PICO_RUNTIME_SKIP_INIT_PSRAM=1 CFG_TUSB_DEBUG=0)
pico_enable_stdio_usb(mic_controller 0)
pico_enable_stdio_uart(mic_controller 0)
pico_add_extra_outputs(mic_controller)
