if(FC_PLAYER_FULL_TILES)
    if(FC_FULL_TILES_WATCHDOG_TEST)
        target_compile_definitions(magicon_standalone_video PRIVATE FC_FULL_TILES_WATCHDOG_TEST=1)
    endif()
    if(FC_FULL_TILES_MOVIE)
        target_compile_definitions(magicon_standalone_video PRIVATE FC_FULL_TILES_MOVIE=1)
    endif()
    if(FC_PLAYER_APU_STREAM OR FC_REMOTE_DESKTOP OR FC_PLAYER_READBACK OR FC_PLAYER_NT_STRESS OR FC_PLAYER_A9_REMAP OR FC_PLAYER_A10_GPIO43)
        message(FATAL_ERROR "FULL_TILES currently requires stock GPIO and the dedicated full-frame driver")
    endif()
    if(NOT FC_FULL_TILES_MOVIE AND NOT EXISTS "${FC_FULL_TILE_IMAGE_HEADER}")
        message(FATAL_ERROR "FULL_TILES requires an external FC_FULL_TILE_IMAGE_HEADER")
    endif()
    if(FC_FULL_TILES_TRACE)
        target_compile_definitions(magicon_standalone_video PRIVATE FC_FULL_TILES_TRACE=1)
    endif()
    set(FC_PLAYER_FAST_CONTROL ON)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(full_driver ${CMAKE_CURRENT_BINARY_DIR}/full_tiles_driver.h)
    set(full_driver_args)
    if(FC_FULL_TILES_MOVIE)
        list(APPEND full_driver_args --movie)
    endif()
    if(FC_PLAYER_NATIVE_PCM)
        list(APPEND full_driver_args --native-pcm)
        if(FC_PLAYER_NATIVE_RAM)
            list(APPEND full_driver_args --ram)
        endif()
    endif()
    add_custom_command(OUTPUT ${full_driver}
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/gen_full_tiles_driver.py ${full_driver} ${full_driver_args}
        DEPENDS gen_full_tiles_driver.py "${CMAKE_CURRENT_LIST_DIR}/../tools/asm6502.py" VERBATIM)
    target_sources(magicon_standalone_video PRIVATE ${full_driver})
    target_include_directories(magicon_standalone_video PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    target_compile_definitions(magicon_standalone_video PRIVATE FC_PLAYER_FULL_TILES=1)
    if(NOT FC_FULL_TILES_MOVIE)
        file(TO_CMAKE_PATH "${FC_FULL_TILE_IMAGE_HEADER}" full_image_header)
        target_compile_definitions(magicon_standalone_video PRIVATE FC_FULL_TILE_IMAGE_HEADER="${full_image_header}")
    endif()
    foreach(name full_tiles_address full_tiles_output)
        pico_generate_pio_header(magicon_standalone_video ${CMAKE_CURRENT_LIST_DIR}/baseline/${name}.pio)
    endforeach()
endif()
