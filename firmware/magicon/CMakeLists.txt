# magicon(USB デバイス = PC 直結)と magicon_wifi(USB ホスト = J6 の ESP32-C6 経由で Wi-Fi)を同じソースから作る。
# magicon_wifi は TinyUSB 0.21 が要るので、build.ps1 の 2 回目のビルド(FC_TUSB021)で作る
function(magicon_target name)
    add_executable(${name} ${CMAKE_CURRENT_LIST_DIR}/main.c ${CMAKE_CURRENT_LIST_DIR}/cart.c ${CMAKE_CURRENT_LIST_DIR}/audio.c)
    if(FC_REMOTE_DMA)
        target_compile_definitions(${name} PRIVATE FC_REMOTE_DMA=1)
    endif()
    if(FC_REMOTE_FULL)
        target_compile_definitions(${name} PRIVATE FC_REMOTE_FULL=1)
    endif()
    pico_generate_pio_header(${name} ${CMAKE_CURRENT_LIST_DIR}/cpu_m2.pio)
    pico_generate_pio_header(${name} ${CMAKE_CURRENT_LIST_DIR}/../chr_test/ppu_chr.pio)
    pico_generate_pio_header(${name} ${CMAKE_CURRENT_LIST_DIR}/mirror.pio)
    pico_generate_pio_header(${name} ${CMAKE_CURRENT_LIST_DIR}/mmc3_a12.pio)
    pico_generate_pio_header(${name} ${CMAKE_CURRENT_LIST_DIR}/rd_width.pio)
    pico_generate_pio_header(${name} ${CMAKE_CURRENT_LIST_DIR}/ppu_constant.pio)
    pico_generate_pio_header(${name} ${CMAKE_CURRENT_LIST_DIR}/ppu_lookup.pio)
    if (FC_DIAG_DMA_CHR)
        target_compile_definitions(${name} PRIVATE FC_DIAG_DMA_CHR=1)
    endif()
    if (FC_DIAG_IMAGE)
        target_compile_definitions(${name} PRIVATE FC_DIAG_IMAGE=1)
    endif()
    if (FC_DIAG_NROM_GAME)
        target_compile_definitions(${name} PRIVATE FC_DIAG_NROM_GAME=1)
    endif()
    if (FC_DIAG_MMC3_GAME)
        target_compile_definitions(${name} PRIVATE FC_DIAG_MMC3_GAME=1)
    endif()
    if (FC_DIAG_MMC3_NO_IRQ)
        target_compile_definitions(${name} PRIVATE FC_DIAG_MMC3_NO_IRQ=1)
    endif()
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../chr_test)   # chr_rom.h(試験画面の PRG)
    target_link_libraries(${name} pico_stdlib pico_multicore hardware_pio hardware_clocks hardware_dma hardware_pwm hardware_watchdog)
    if (FC_DIAGNOSTIC)
        target_compile_definitions(${name} PRIVATE FC_DIAGNOSTIC=1)
    endif()
    if (FC_DIAG_CPU_ONLY)
        target_compile_definitions(${name} PRIVATE FC_DIAG_CPU_ONLY=1)
    endif()
    if (FC_DIAG_SPLIT)
        target_compile_definitions(${name} PRIVATE FC_DIAG_SPLIT=1)
    endif()
    if (FC_DIAG_CONST_CHR)
        target_compile_definitions(${name} PRIVATE FC_DIAG_CONST_CHR=1)
    endif()
    pico_enable_stdio_uart(${name} 0)
    if (FC_DIAG_LIVE)
        target_compile_definitions(${name} PRIVATE FC_DIAG_LIVE=1)
    endif()
    pico_add_extra_outputs(${name})
    if (FC_DIAG_DIRECT_CHR)
        target_compile_definitions(${name} PRIVATE FC_DIAG_DIRECT_CHR=1)
    endif()
endfunction()

if (FC_TUSB021)
    magicon_target(magicon_wifi)
    target_compile_definitions(magicon_wifi PRIVATE MAGICON_USB_HOST=1)
    target_include_directories(magicon_wifi PRIVATE ${CMAKE_CURRENT_LIST_DIR}/usb_host)   # tusb_config.h(ホスト)
    target_link_libraries(magicon_wifi tinyusb_host)
    pico_enable_stdio_usb(magicon_wifi 0)     # USB はホストに使う
else()
    magicon_target(magicon)
    pico_enable_stdio_usb(magicon 1)
endif()
